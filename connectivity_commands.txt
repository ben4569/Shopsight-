Python Connectivity Script:

import os
import mysql.connector

DB = {
    "host": os.getenv("MYSQL_HOST", "127.0.0.1"),
    "port": 3306,
    "user": os.getenv("MYSQL_USER", "root"),
    "password": os.getenv("MYSQL_PASSWORD", ""),
    "database": "shopsight"
}

def db():
    return mysql.connector.connect(**DB)

def setup():
    c = mysql.connector.connect(
        host=DB["host"],
        port=DB["port"],
        user=DB["user"],
        password=DB["password"]
    )
    q = c.cursor()
    q.execute("CREATE DATABASE IF NOT EXISTS shopsight")
    q.close()
    c.close()

    c = db()
    q = c.cursor()
    q.execute("""CREATE TABLE IF NOT EXISTS users(
        id INT AUTO_INCREMENT PRIMARY KEY,
        name VARCHAR(100),
        email VARCHAR(200) UNIQUE,
        password VARCHAR(255))""")

    q.execute("""CREATE TABLE IF NOT EXISTS products(
        id INT AUTO_INCREMENT PRIMARY KEY,
        user_id INT,
        name VARCHAR(100),
        cost FLOAT,
        price FLOAT,
        stock INT)""")

    q.execute("""CREATE TABLE IF NOT EXISTS sales(
        id INT AUTO_INCREMENT PRIMARY KEY,
        user_id INT,
        product_id INT,
        units INT,
        price FLOAT,
        cost FLOAT,
        sale_date DATE)""")

    c.commit()
    q.close()
    c.close()
